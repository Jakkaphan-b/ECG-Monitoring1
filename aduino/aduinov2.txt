// ฟังก์ชันตรวจจับ R-peak และคำนวณ Heart Rate
void detectRPeakAndCalculateHR(int ecgValue) {
  static int previousValue = 0;
  static int previousPreviousValue = 0;
  unsigned long currentTime = millis();
  
  // ตรวจจับ R-peak (จุดสูงสุดที่มีค่าเกิน threshold)
  if (ecgValue > ECG_THRESHOLD && 
      ecgValue > previousValue && 
      previousValue > previousPreviousValue && 
      !peakDetected &&
      (currentTime - lastPeakTime) > MIN_RR_INTERVAL) {
    
    peakDetected = true;
    lastPeakValue = ecgValue;
    
    if (lastPeakTime > 0) {
      // คำนวณ RR interval
      unsigned long rrInterval = currentTime - lastPeakTime;
      
      // ตรวจสอบว่า RR interval อยู่ในช่วงที่สมเหตุสมผล
      if (rrInterval >= MIN_RR_INTERVAL && rrInterval <= MAX_RR_INTERVAL) {
        lastRRInterval = rrInterval;
        
        // คำนวณ heart rate จาก RR interval
        int instantHR = 60000 / rrInterval;
        
        // เก็บค่าใน buffer สำหรับการเฉลี่ย
        hrBuffer[hrIndex % HR_WINDOW_SIZE] = instantHR;
        hrIndex++;
        
        // คำนวณค่าเฉลี่ย heart rate
        int sum = 0;
        int count = min(hrIndex, HR_WINDOW_SIZE);
        for (int i = 0; i < count; i++) {
          sum += hrBuffer[i];
        }
        currentHeartRate = sum / count;
        
        Serial.println("💓 R-peak detected! RR: " + String(rrInterval) + 
                      "ms, Instant HR: " + String(instantHR) + 
                      " BPM, Avg HR: " + String(currentHeartRate) + " BPM");
      }
    }
    
    lastPeakTime = currentTime;
  }
  
  // Reset peak detection หลังจาผ่านไป 50ms
  if (peakDetected && (currentTime - lastPeakTime) > 50) {
    peakDetected = false;
  }
  
  // เก็บค่าก่อนหน้าสำหรับการตรวจจับ peak
  previousPreviousValue = previousValue;
  previousValue = ecgValue;
}

// ฟังก์ชันดึงค่า heart rate ปัจจุบัน
int getCurrentHeartRate() {
  // หากไม่มี peak ใหม่มานานเกิน 3 วินาที ให้ return 0
  if (millis() - lastPeakTime > 3000) {
    return 0;
  }
  return currentHeartRate;
}

// ฟังก์ชันแสดงผลบน OLED
void updateOLED() {
  display.clearDisplay();

  // 1. ชื่อ Device (ซ้ายบน)
  display.setTextSize(1);
  display.setTextColor(1);
  display.setCursor(0, 0);
  display.print("ECG_001");

  // 2. อัตราการเต้นหัวใจ (กลางบน)
  int heartRate = getCurrentHeartRate();
  display.setCursor(50, 0);
  if (heartRate > 0) {
    display.setTextSize(1);
    display.print(heartRate);
    display.print(" BPM");
  } else {
    display.setTextSize(1);
    display.print("-- BPM");
  }

  // 3. สัญลักษณ์ WiFi (ขวาบน) - แสดงสถานะแม้ offline
  display.setCursor(110, 0);
  display.setTextSize(1);
  if (WiFi.status() == WL_CONNECTED) {
    display.print("WiFi");
  } else {
    display.print("X");
  }

  // 4. กราฟ ECG (พื้นที่ด้านล่าง) - แสดงแม้ไม่มี WiFi
  int graphHeight = 45;
  int graphTop = 18;
  int graphLeft = 0;
  int graphWidth = 128;
  
  // วาดเส้นแบ่งระหว่างส่วนบนและกราฟ
  display.drawLine(0, 16, 128, 16, 1);
  
  // วาดกราฟ ECG จากข้อมูลใน buffer
  static int ecgBuffer[128];
  static int bufferIndex = 0;
  
  // เพิ่มค่า ECG ล่าสุดลงใน buffer สำหรับแสดงผล
  if (idxBuf > 0) {
    ecgBuffer[bufferIndex] = buf[idxBuf - 1];
    bufferIndex = (bufferIndex + 1) % 128;
  }
  
  // วาดกราฟ ECG แบบ scrolling
  for (int i = 1; i < 128; i++) {
    int prevIndex = (bufferIndex + i - 1) % 128;
    int currIndex = (bufferIndex + i) % 128;
    
    int y1 = graphTop + graphHeight - map(ecgBuffer[prevIndex], 0, 4095, 0, graphHeight);
    int y2 = graphTop + graphHeight - map(ecgBuffer[currIndex], 0, 4095, 0, graphHeight);
    
    // จำกัดค่า y ไม่ให้เกินขอบเขต
    y1 = constrain(y1, graphTop, graphTop + graphHeight);
    y2 = constrain(y2, graphTop, graphTop + graphHeight);
    
    display.drawLine(i - 1, y1, i, y2, 1);
  }

  // แสดงข้อมูลเพิ่มเติมที่มุมล่าง
  display.setCursor(0, 56);
  display.setTextSize(1);
  if (lastRRInterval > 0) {
    display.print("RR:");
    display.print(lastRRInterval);
    display.print("ms");
  } else {
    // แสดงสถานะเมื่อไม่มีการตรวจจับ heart rate
    display.print("Waiting signal...");
  }

  // แสดงสถานะ WiFi ที่มุมขวาล่าง
  display.setCursor(90, 56);
  if (WiFi.status() == WL_CONNECTED) {
    display.print("Online");
  } else {
    display.print("Offline");
  }

  display.display();
}

bool loadConfigFromSPIFFS(String &ssid, String &pass) {
  if (!SPIFFS.begin(true)) { // true = format if mount fails
    Serial.println("SPIFFS mount failed, formatting...");
    if (!SPIFFS.begin(true)) {
      Serial.println("SPIFFS format failed!");
      return false;
    }
    Serial.println("SPIFFS formatted successfully");
  }
  
  if (!SPIFFS.exists(WIFI_CONFIG_PATH)) {
    Serial.println("WiFi config file not found");
    return false;
  }
  
  File f = SPIFFS.open(WIFI_CONFIG_PATH, "r");
  if (!f) { 
    Serial.println("Failed to open config file"); 
    return false; 
  }
  String s;
  while (f.available()) s += (char)f.read();
  f.close();
  
  Serial.println("Config file content: " + s);
  
  // ใช้ ArduinoJson ในการ parse เพื่อความแม่นยำ
  DynamicJsonDocument doc(512);
  DeserializationError error = deserializeJson(doc, s);
  if (error) {
    Serial.println("Failed to parse config file: " + String(error.c_str()));
    return false;
  }
  
  if (!doc.containsKey("ssid") || !doc.containsKey("pass")) {
    Serial.println("Config file missing ssid or pass");
    return false;
  }
  
  ssid = String(doc["ssid"].as<const char*>());
  pass = String(doc["pass"].as<const char*>());
  
  Serial.println("Loaded SSID: '" + ssid + "'");
  Serial.println("Loaded Pass length: " + String(pass.length()));
  
  // ตรวจสอบว่า SSID ไม่ว่าง
  if (ssid.length() == 0) {
    Serial.println("SSID is empty!");
    return false;
  }
  
  return true;
}

void saveConfigToSPIFFS(const String &ssid, const String &pass) {
  Serial.println("Attempting to save WiFi credentials to SPIFFS...");
  Serial.println("SSID: '" + ssid + "'");
  Serial.println("Pass length: " + String(pass.length()));
  
  if (!SPIFFS.begin(true)) { // true = format if mount fails
    Serial.println("SPIFFS mount failed for saving, formatting...");
    if (!SPIFFS.begin(true)) {
      Serial.println("SPIFFS format failed!");
      return;
    }
    Serial.println("SPIFFS formatted successfully");
  }
  
  // ใช้ ArduinoJson สำหรับการสร้าง JSON ที่ถูกต้อง
  DynamicJsonDocument doc(512);
  doc["ssid"] = ssid;
  doc["pass"] = pass;
  
  File f = SPIFFS.open(WIFI_CONFIG_PATH, "w");
  if (!f) { 
    Serial.println("Failed to open config file for writing"); 
    return; 
  }
  
  // เขียน JSON ลงไฟล์
  size_t bytes_written = serializeJson(doc, f);
  f.close();
  
  if (bytes_written > 0) {
    Serial.println("Successfully saved " + String(bytes_written) + " bytes to SPIFFS");
    
    // อ่านกลับมาเพื่อยืนยัน
    String verification;
    File verify_f = SPIFFS.open(WIFI_CONFIG_PATH, "r");
    if (verify_f) {
      while (verify_f.available()) verification += (char)verify_f.read();
      verify_f.close();
      Serial.println("Verification read: " + verification);
    }
  } else {
    Serial.println("Failed to write to SPIFFS file");
  }
}

  configServer.send(200, "text/html", config_html);
}

void handleScan() {
  String json = "[";
  int n = WiFi.scanNetworks();
  for (int i = 0; i < n; i++) {
    String ssid = WiFi.SSID(i);
    int rssi = WiFi.RSSI(i);
    bool encrypted = (WiFi.encryptionType(i) != WIFI_AUTH_OPEN);
    
    json += "{";
    json += "\"s\":\"" + ssid + "\",";
    json += "\"r\":" + String(rssi) + ",";
    json += "\"e\":" + String(encrypted ? 1 : 0);
    json += "}";
    if (i < n - 1) json += ",";
  }
  json += "]";
  
  configServer.send(200, "application/json", json);
}

void handleSave() {
  String ssid = configServer.arg("ssid");
  String pass = configServer.arg("pass");
  
  // ตรวจสอบข้อมูลที่ได้รับ
  Serial.println("=== Handling Save Request ===");
  Serial.println("Received SSID: '" + ssid + "'");
  Serial.println("Password length: " + String(pass.length()));
  
  if (ssid.length() == 0) {
    Serial.println("Error: SSID is empty");
    configServer.send(400, "text/plain", "Missing SSID");
    return;
  }
  
  // ลบช่องว่างหน้าและหลัง SSID/Password
  ssid.trim();
  pass.trim();
  
  Serial.println("Trimmed SSID: '" + ssid + "'");
  Serial.println("Trimmed Password length: " + String(pass.length()));

  WiFi.disconnect(true); // ตัดการเชื่อมต่อเก่าก่อน

    delay(500);
    Serial.print(".");

    Serial.println("║           WiFi Connection Success!    ║");
    Serial.println("╠═══════════════════════════════════════╣");

    String ssid = WiFi.SSID();

    for (int i = ssid.length(); i < 20; i++) Serial.print(" ");
    Serial.println("║");
    Serial.print("║ 🌐 IP Address: ");
    String ip = WiFi.localIP().toString();
    Serial.print(ip);
    for (int i = ip.length(); i < 17; i++) Serial.print(" ");
    Serial.println("║");
    Serial.print("║ 🚪 Gateway: ");
    String gw = WiFi.gatewayIP().toString();
    Serial.print(gw);
    for (int i = gw.length(); i < 20; i++) Serial.print(" ");
    Serial.println("║");
    Serial.print("║ 🔗 Subnet Mask: ");
    String subnet = WiFi.subnetMask().toString();
    Serial.print(subnet);
    for (int i = subnet.length(); i < 16; i++) Serial.print(" ");
    Serial.println("║");
    Serial.print("║ 📡 DNS Server: ");
    String dns = WiFi.dnsIP().toString();

void startConfigPortal() {
  Serial.println("Starting ECG Device Config Portal...");
  
  // Start WiFi in AP mode
  WiFi.mode(WIFI_AP);
  WiFi.softAP("ECG_Device");
  
  IPAddress apIP = WiFi.softAPIP();
  Serial.println("=== Config Portal Information ===");
  Serial.println("Access Point: ECG_Device");
  Serial.print("Portal IP: ");
  Serial.println(apIP);
  Serial.println("Connect to 'ECG_Device' and open: http://" + apIP.toString());
  Serial.println("================================");
  
  // Setup web server routes
  configServer.on("/", handleRoot);
  configServer.on("/scan", handleScan);
  configServer.on("/save", HTTP_POST, handleSave);
  
  configServer.begin();
  Serial.println("HTTP server started successfully");
}

void setupWiFi() {

  // Try to load saved credentials first
  bool have_config = loadConfigFromSPIFFS(wifi_ssid, wifi_pass);
  
  if (have_config && wifi_ssid.length() > 0) {
    Serial.println("Found valid saved WiFi credentials, attempting connection...");
    Serial.println("SSID: '" + wifi_ssid + "'");
    Serial.println("Password length: " + String(wifi_pass.length()));
    
    // ตั้งค่า WiFi persistence และ auto-reconnect
    WiFi.persistent(true);
    WiFi.setAutoReconnect(true);
    WiFi.mode(WIFI_STA);
    
    // ตัดการเชื่อมต่อเก่าก่อน

      wl_status_t status = WiFi.status();
      if (status == WL_CONNECTED) {
        Serial.println("\n╔═══════════════════════════════════════╗");
        Serial.println("║        🎯 WiFi Connected Successfully! ║");
        Serial.println("╠═══════════════════════════════════════╣");
        Serial.print("║ 📶 Network: ");
        String ssid = WiFi.SSID();
        Serial.print(ssid);
        for (int i = ssid.length(); i < 22; i++) Serial.print(" ");
        Serial.println("║");
        Serial.print("║ 🌐 IP Address: ");
        String ip = WiFi.localIP().toString();
        Serial.print(ip);
        for (int i = ip.length(); i < 19; i++) Serial.print(" ");
        Serial.println("║");
        Serial.print("║ 🚪 Gateway: ");
        String gw = WiFi.gatewayIP().toString();
        Serial.print(gw);
        for (int i = gw.length(); i < 22; i++) Serial.print(" ");
        Serial.println("║");
        Serial.print("║ 🔗 Subnet: ");
        String subnet = WiFi.subnetMask().toString();
        Serial.print(subnet);
        for (int i = subnet.length(); i < 23; i++) Serial.print(" ");
        Serial.println("║");
        Serial.print("║ 📡 DNS: ");
        String dns = WiFi.dnsIP().toString();
        Serial.print(dns);
        for (int i = dns.length(); i < 26; i++) Serial.print(" ");
        Serial.println("║");
        Serial.print("║ 📊 Signal: ");
        int rssi = WiFi.RSSI();
        Serial.print(rssi);
        Serial.print(" dBm (");
        if (rssi > -50) Serial.print("Excellent");
        else if (rssi > -60) Serial.print("Good");
        else if (rssi > -70) Serial.print("Fair");
        else Serial.print("Weak");
        Serial.print(")");
        String quality = (rssi > -50) ? "Excellent" : (rssi > -60) ? "Good" : (rssi > -70) ? "Fair" : "Weak";
        String rssi_line = String(rssi) + " dBm (" + quality + ")";
        for (int i = rssi_line.length(); i < 21; i++) Serial.print(" ");
        Serial.println("║");
        Serial.print("║ 🏷️  MAC: ");
        String mac = WiFi.macAddress();
        Serial.print(mac);
        for (int i = mac.length(); i < 26; i++) Serial.print(" ");
        Serial.println("║");
        Serial.println("╚═══════════════════════════════════════╝");
        return;
      }
      
      // แสดงสถานะการเชื่อมต่อ
      if ((millis() - start) % 2000 == 0) {
        Serial.println("");
        Serial.println("WiFi Status: " + String(status));
        Serial.print("Connecting");
      }
      
      delay(500);
      Serial.print(".");
    }
    
    Serial.println("");

  } else {

  }

  startConfigPortal();
  
  // Keep the portal running until connected
  while (WiFi.status() != WL_CONNECTED) {
    configServer.handleClient();
    delay(10);
  }
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  Serial.println("\n=== ECG Monitor Starting ===");
  Serial.println("Firmware: v1.0.1");
  Serial.println("Device: ECG_001");
  Serial.println("==========================");

  // เริ่มต้น OLED Display
  Wire.begin(OLED_SDA, OLED_SCL);
  if (!display.begin(OLED_ADDR, true)) {
    Serial.println("SH1106 allocation failed");
  }else {
    Serial.println("OLED initialized successfully");
    // แสดงหน้า Normal Operation ทันที
    display.clearDisplay();
    display.setTextSize(1);
    display.setTextColor(1);
    display.setCursor(0, 0);
    display.print("ECG_001");
    display.setCursor(50, 0);
    display.print("-- BPM");
    display.setCursor(110, 0);
    display.print("X");
    display.drawLine(0, 16, 128, 16, 1);
    display.setCursor(0, 56);
    display.print("Initializing...");
    display.display();
  }

  analogReadResolution(12);                    // 0..4095
  analogSetPinAttenuation(ECG_PIN, ADC_11db);  // ~3.3V

  // Setup WiFi using custom interface
  setupWiFi();

  // ตั้งค่าเวลา NTP สำหรับ timestamp ที่ถูกต้อง
  Serial.println("🕐 Setting up NTP time...");
  configTime(7 * 3600, 0, "pool.ntp.org", "time.nist.gov"); // GMT+7 for Thailand
  
  Serial.print("Waiting for NTP time sync");
  time_t now = time(nullptr);
  int sync_attempts = 0;
  while (now < 100000 && sync_attempts < 30) {
    delay(500);
    Serial.print(".");
    now = time(nullptr);
    sync_attempts++;
  }
  Serial.println("");
  
  if (now >= 100000) {
    unsigned long timestamp_ms = (unsigned long)now * 1000;
    Serial.println("✅ NTP time synchronized successfully!");
    Serial.println("📅 Current time: " + String(ctime(&now)));
    Serial.println("🔢 Timestamp (ms): " + String(timestamp_ms));
  } else {
    Serial.println("⚠️  NTP sync failed, using system time");
  }

  // ทดสอบง่าย ๆ (ไม่ตรวจ CA) — สำหรับ Production ควรใส่ root CA จริง
  secureClient.setInsecure();
  
  Serial.println("=== ECG Monitoring Started ===");
  Serial.println("Sampling Rate: " + String(SAMPLE_RATE) + " Hz");
  Serial.println("Chunk Size: " + String(CHUNK_SIZE) + " samples");
  Serial.println("ECG Pin: " + String(ECG_PIN));
  Serial.println("Firebase Host: " + String(RTDB_HOST));
  Serial.println("💓 Heart Rate Detection: ENABLED");
  Serial.println("🎯 R-peak Threshold: " + String(ECG_THRESHOLD));
  Serial.println("📊 HR Window Size: " + String(HR_WINDOW_SIZE) + " samples");
  Serial.println("⚡ RR Range: " + String(MIN_RR_INTERVAL) + "-" + String(MAX_RR_INTERVAL) + " ms");
  Serial.println("📺 OLED Display: SH1106 1.3\" (I2C: " + String(OLED_SDA) + "/" + String(OLED_SCL) + ")");
  Serial.println("=============================");
}

void loop() {
  // Check WiFi connection and reconnect if needed
  static unsigned long lastWiFiCheck = 0;
  static bool wasConnected = false;
  
  if (millis() - lastWiFiCheck > 1000) { // ตรวจสอบทุกวินาที
    bool isConnected = (WiFi.status() == WL_CONNECTED);
    
    if (!isConnected && wasConnected) {
      Serial.println("⚠️  WiFi connection lost! Status: " + String(WiFi.status()));
    } else if (isConnected && !wasConnected) {
      Serial.println("✅ WiFi reconnected!");
    }
    
    if (!isConnected) {
      Serial.println("WiFi disconnected, attempting reconnect...");
      setupWiFi();
      return;
    }
    
    wasConnected = isConnected;
    lastWiFiCheck = millis();
  }
  
  int v = analogRead(ECG_PIN);  // 0–4095
  buf[idxBuf++] = v;

  // ตรวจจับ R-peak และคำนวณ Heart Rate
  detectRPeakAndCalculateHR(v);

  // อัปเดต OLED Display ทุก 100ms เพื่อให้ smooth มากขึ้น
  static unsigned long lastOLEDUpdate = 0;
  if (millis() - lastOLEDUpdate > 100) {
    updateOLED();
    lastOLEDUpdate = millis();
  }

  if (idxBuf >= CHUNK_SIZE) {
    sendChunk(buf, CHUNK_SIZE);
    idxBuf = 0;
  }

  if (millis() - lastStatus > 5000) {
    sendStatus();
    lastStatus = millis();
    
    // แสดงสถานะ WiFi ทุก 30 วินาที
    static unsigned long lastWiFiStatus = 0;
    if (millis() - lastWiFiStatus > 30000) {
      showWiFiStatus();
      lastWiFiStatus = millis();
    }
    
    // Debug timestamp และ heart rate ทุก 60 วินาที
    static unsigned long lastTimeDebug = 0;
    if (millis() - lastTimeDebug > 60000) {
      Serial.println("=== System Status Debug ===");
      printCurrentTime();
      Serial.println("💓 Current Heart Rate: " + String(getCurrentHeartRate()) + " BPM");
      Serial.println("⏱️  Last RR Interval: " + String(lastRRInterval) + " ms");
      Serial.println("📡 Last Peak Time: " + String(millis() - lastPeakTime) + " ms ago");
      Serial.println("===========================");
      lastTimeDebug = millis();
    }
  }

  delayMicroseconds(1000000UL / SAMPLE_RATE);  // 250 Hz ~ 4000 µs
}

void sendChunk(int* data, int n) {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  // ใช้ secure client
  if (!http.begin(secureClient, chunksUrl())) {
    Serial.println("begin() failed");
    return;
  }
  http.addHeader("Content-Type", "application/json");

  DynamicJsonDocument doc(2048);
  doc["sampleRate"] = SAMPLE_RATE;
  
  // ใช้ Unix timestamp (milliseconds) สำหรับ ts
  time_t now = time(nullptr);
  unsigned long timestamp_ms = (unsigned long)now * 1000;
  doc["ts"] = timestamp_ms;
  
  JsonArray arr = doc.createNestedArray("values");
  for (int i = 0; i < n; i++) arr.add(data[i]);

  String body; serializeJson(doc, body);
  int code = http.POST(body);

  Serial.printf("POST chunk %d vals -> %d\n", n, code);
  if (code > 0 && code != 200) {
    Serial.println(http.getString());
  }
  http.end();
}

void sendStatus() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  if (!http.begin(secureClient, statusUrl())) {
    Serial.println("begin() failed");
    return;
  }
  http.addHeader("Content-Type", "application/json");

  DynamicJsonDocument doc(512);
  doc["connected"] = true;
  
  // ใช้ Unix timestamp (milliseconds) แทน millis() สำหรับ last_seen
  time_t now = time(nullptr);
  unsigned long timestamp_ms = (unsigned long)now * 1000;
  doc["last_seen"] = timestamp_ms;
  
  doc["rssi"] = WiFi.RSSI();            // dBm
  doc["battery"] = nullptr;             // ถ้ามีวงจรวัดแบต ให้ใส่ %
  doc["firmware_version"] = "1.0.1";    // อัปเดตเวอร์ชัน
  
  // เพิ่มค่า Heart Rate
  int heartRate = getCurrentHeartRate();
  doc["heart_rate"] = heartRate;
  doc["rr_interval"] = lastRRInterval;
  
  // เพิ่ม debug info
  Serial.println("Current timestamp: " + String(timestamp_ms) + " (" + String(ctime(&now)) + ")");
  Serial.println("Sending Heart Rate: " + String(heartRate) + " BPM, RR: " + String(lastRRInterval) + " ms");

  String body; serializeJson(doc, body);
  int code = http.PUT(body);

  Serial.printf("PUT status -> %d\n", code);
  if (code > 0 && code != 200) {
    Serial.println(http.getString());
  }
  http.end();
}
